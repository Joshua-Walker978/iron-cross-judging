src.hold_detection

Detect the held part of an iron cross from a per-frame arm-angle signal.

The input is a table with one row per video frame (normally produced by src.pose_estimation.extract_deviation_series) holding the deviation from horizontal of each arm. The processing chain is:

  1. smooth_deviations - fill missed detections by linear interpolation, then apply a Savitzky-Golay filter to suppress landmark jitter.
  2. add_angular_velocity - central-difference derivative of the smoothed angles with respect to time.
  3. flag_stopped_frames - a frame is "stopped" when both arms move slower than a velocity threshold.
  4. find_stopped_segments - group consecutive stopped frames.
  5. merge_segments - join segments separated by short gaps (brief tremors or detection noise during a real hold).
  6. build_candidate_holds and select_main_hold - keep segments whose mean angle could plausibly be a cross and take the longest one.

detect_hold runs the whole chain, and judge_hold converts the result to deduction categories.

Default parameter values were chosen by inspecting the HPCComp.mov recording at about 30 frames per second; see the module constants.

 59def choose_window_length(n_frames, preferred=DEFAULT_WINDOW_LENGTH, minimum=MIN_WINDOW_LENGTH):
 60    """
 61    Pick an odd Savitzky-Golay window that fits the clip length.
 62
 63    Parameters
 64    ----------
 65    n_frames : int
 66        Number of frames in the clip.
 67    preferred : int, optional
 68        Window to use when the clip is long enough. Default 9.
 69    minimum : int, optional
 70        Smallest acceptable window. Default 5.
 71
 72    Returns
 73    -------
 74    int
 75        ``min(preferred, largest odd number <= n_frames)``.
 76
 77    Raises
 78    ------
 79    ValueError
 80        If `n_frames` is smaller than `minimum`.
 81
 82    Examples
 83    --------
 84    >>> choose_window_length(141)
 85    9
 86    >>> choose_window_length(8)
 87    7
 88    """
 89    if n_frames < minimum:
 90        raise ValueError(f"Need at least {minimum} frames to smooth, got {n_frames}")
 91    largest_odd = n_frames if n_frames % 2 == 1 else n_frames - 1
 92    return max(min(preferred, largest_odd), minimum)

102def smooth_deviations(df, window_length=DEFAULT_WINDOW_LENGTH, polyorder=DEFAULT_POLYORDER):
103    """
104    Interpolate missed detections and smooth both arm-angle signals.
105
106    Parameters
107    ----------
108    df : pandas.DataFrame
109        One row per frame with columns ``left_deviation`` and
110        ``right_deviation`` in degrees. Frames with no detected pose hold
111        NaN.
112    window_length : int, optional
113        Savitzky-Golay window in frames; must be odd and no longer than the
114        clip. Default 9.
115    polyorder : int, optional
116        Polynomial order; must be less than `window_length`. Default 2.
117
118    Returns
119    -------
120    pandas.DataFrame
121        A copy of `df` with added columns ``left_smooth`` and
122        ``right_smooth`` (degrees).
123
124    Raises
125    ------
126    ValueError
127        If an arm has no detected value in any frame.
128
129    Notes
130    -----
131    Interior gaps are filled by linear interpolation between the
132    neighbouring detected frames. Gaps at the start or end of the clip are
133    filled with the nearest detected value; without this the Savitzky-Golay
134    edge fit receives NaN and SciPy raises an error.
135    """
136    _check_columns(df, ("left_deviation", "right_deviation"))
137    out = df.copy()
138    for side in ("left", "right"):
139        filled = out[f"{side}_deviation"].astype(float).interpolate(limit_direction="both")
140        if filled.isna().all():
141            raise ValueError(f"No {side}-arm angle was detected in any frame")
142        out[f"{side}_smooth"] = savgol_filter(filled.to_numpy(), window_length, polyorder)
143    return out

146def add_angular_velocity(df):
147    """
148    Add the angular velocity of each arm, in degrees per second.
149
150    Parameters
151    ----------
152    df : pandas.DataFrame
153        Output of `smooth_deviations`, with columns ``time_sec``,
154        ``left_smooth`` and ``right_smooth``.
155
156    Returns
157    -------
158    pandas.DataFrame
159        A copy of `df` with added columns ``left_velocity`` and
160        ``right_velocity`` (deg/s).
161
162    Notes
163    -----
164    Uses `numpy.gradient`: second-order central differences inside the
165    clip and first-order one-sided differences at the two ends, evaluated
166    against the actual frame timestamps.
167    """
168    _check_columns(df, ("time_sec", "left_smooth", "right_smooth"))
169    out = df.copy()
170    t = out["time_sec"].to_numpy(dtype=float)
171    for side in ("left", "right"):
172        out[f"{side}_velocity"] = np.gradient(out[f"{side}_smooth"].to_numpy(dtype=float), t)
173    return out

176def flag_stopped_frames(df, velocity_threshold=VELOCITY_THRESHOLD_DEG_S):
177    """
178    Mark frames where both arms are effectively stationary.
179
180    Parameters
181    ----------
182    df : pandas.DataFrame
183        Output of `add_angular_velocity`.
184    velocity_threshold : float, optional
185        Speed limit in deg/s that both arms must be under. Default 15.
186
187    Returns
188    -------
189    pandas.DataFrame
190        A copy of `df` with a boolean column ``is_stopped``. Frames with a
191        NaN velocity are never stopped.
192    """
193    _check_columns(df, ("left_velocity", "right_velocity"))
194    out = df.copy()
195    out["is_stopped"] = (out["left_velocity"].abs() < velocity_threshold) & (
196        out["right_velocity"].abs() < velocity_threshold
197    )
198    return out

201def find_stopped_segments(df):
202    """
203    Group consecutive stopped frames into segments.
204
205    Parameters
206    ----------
207    df : pandas.DataFrame
208        Output of `flag_stopped_frames`, with columns ``time_sec``,
209        ``is_stopped``, ``left_smooth`` and ``right_smooth``.
210
211    Returns
212    -------
213    pandas.DataFrame
214        One row per run of stopped frames, in time order, with columns
215        ``start_time`` and ``end_time`` (timestamps of the first and last
216        frame, s), ``avg_left_angle`` and ``avg_right_angle`` (deg),
217        ``n_frames`` and ``duration`` (``end_time - start_time``, s; a
218        single-frame segment has duration 0). Empty if no frame is stopped.
219    """
220    _check_columns(df, ("time_sec", "is_stopped", "left_smooth", "right_smooth"))
221    stopped_mask = df["is_stopped"].astype(bool)
222    segment_id = (stopped_mask != stopped_mask.shift()).cumsum()
223    stopped = df[stopped_mask]
224    segments = stopped.groupby(segment_id[stopped_mask]).agg(
225        start_time=("time_sec", "min"),
226        end_time=("time_sec", "max"),
227        avg_left_angle=("left_smooth", "mean"),
228        avg_right_angle=("right_smooth", "mean"),
229        n_frames=("time_sec", "count"),
230    )
231    segments["duration"] = segments["end_time"] - segments["start_time"]
232    return segments.reset_index(drop=True)

235def merge_segments(segments, gap_tolerance=GAP_TOLERANCE_S):
236    """
237    Merge stopped segments separated by short gaps.
238
239    Parameters
240    ----------
241    segments : pandas.DataFrame
242        Output of `find_stopped_segments`, sorted by ``start_time``.
243    gap_tolerance : float, optional
244        Largest gap in seconds, from the end of one segment to the start of
245        the next, that is bridged. Default 0.25.
246
247    Returns
248    -------
249    list of tuple of float
250        ``(start_time, end_time)`` pairs in seconds, in time order. Empty if
251        `segments` is empty.
252
253    Examples
254    --------
255    >>> segs = pd.DataFrame({"start_time": [0.0, 1.1, 3.0], "end_time": [1.0, 2.0, 3.5]})
256    >>> merge_segments(segs, gap_tolerance=0.25)
257    [(0.0, 2.0), (3.0, 3.5)]
258    """
259    merged = []
260    cur_start = cur_end = None
261    for _, row in segments.iterrows():
262        if cur_start is None:
263            cur_start, cur_end = row["start_time"], row["end_time"]
264        elif row["start_time"] - cur_end <= gap_tolerance:
265            cur_end = row["end_time"]
266        else:
267            merged.append((float(cur_start), float(cur_end)))
268            cur_start, cur_end = row["start_time"], row["end_time"]
269    if cur_start is not None:
270        merged.append((float(cur_start), float(cur_end)))
271    return merged

274def average_angle_in_segment(df, start, end):
275    """
276    Mean deviation over both arms between two timestamps (inclusive).
277
278    Parameters
279    ----------
280    df : pandas.DataFrame
281        Frame table with ``time_sec``, ``left_smooth`` and ``right_smooth``.
282    start, end : float
283        Segment limits in seconds.
284
285    Returns
286    -------
287    float
288        Mean of the left-arm and right-arm mean smoothed deviations, in
289        degrees. NaN if no frame falls in the interval.
290    """
291    mask = (df["time_sec"] >= start) & (df["time_sec"] <= end)
292    return float(df.loc[mask, ["left_smooth", "right_smooth"]].mean().mean())

295def build_candidate_holds(df, merged_segments):
296    """
297    Describe each merged segment as a candidate hold.
298
299    Parameters
300    ----------
301    df : pandas.DataFrame
302        Frame table with ``time_sec``, ``left_smooth`` and ``right_smooth``.
303    merged_segments : list of tuple of float
304        Output of `merge_segments`.
305
306    Returns
307    -------
308    pandas.DataFrame
309        Columns ``start``, ``end``, ``duration`` (s) and ``avg_angle``
310        (deg), one row per segment.
311    """
312    rows = [
313        {
314            "start": start,
315            "end": end,
316            "duration": end - start,
317            "avg_angle": average_angle_in_segment(df, start, end),
318        }
319        for start, end in merged_segments
320    ]
321    return pd.DataFrame(rows, columns=["start", "end", "duration", "avg_angle"])

324def select_main_hold(candidates, max_angle=MAX_CROSS_ANGLE_DEG):
325    """
326    Choose the hold to judge from the candidate segments.
327
328    Parameters
329    ----------
330    candidates : pandas.DataFrame
331        Output of `build_candidate_holds`.
332    max_angle : float, optional
333        Candidates with a mean deviation above this (deg) are discarded as
334        not being a cross position, e.g. hanging with arms overhead.
335        Default 45.
336
337    Returns
338    -------
339    pandas.Series or None
340        The longest remaining candidate, with fields ``start``, ``end``,
341        ``duration`` and ``avg_angle``; None if no candidate qualifies.
342    """
343    if candidates.empty:
344        return None
345    plausible = candidates[candidates["avg_angle"] <= max_angle]
346    if plausible.empty:
347        return None
348    return plausible.loc[plausible["duration"].idxmax()]

351def classify_hold_duration(duration_sec):
352    """
353    Classify how long the hold lasted.
354
355    Parameters
356    ----------
357    duration_sec : float
358        Hold duration in seconds.
359
360    Returns
361    -------
362    str
363        ``"No deduction"`` for at least 2 s, ``"Medium error"`` for a stop
364        shorter than 2 s, and ``"Large error and non-recognition (no stop)"``
365        for zero duration.
366
367    Raises
368    ------
369    ValueError
370        If `duration_sec` is NaN or negative.
371
372    Examples
373    --------
374    >>> classify_hold_duration(1.96)
375    'Medium error'
376    """
377    if duration_sec is None or np.isnan(duration_sec):
378        raise ValueError("duration_sec is NaN or None")
379    if duration_sec < 0:
380        raise ValueError(f"duration_sec must be >= 0, got {duration_sec}")
381
382    if duration_sec >= MIN_HOLD_DURATION_S:
383        return "No deduction"
384    elif duration_sec > 0:
385        return "Medium error"
386    else:
387        return "Large error and non-recognition (no stop)"

390def detect_hold(
391    series,
392    window_length=None,
393    polyorder=DEFAULT_POLYORDER,
394    velocity_threshold=VELOCITY_THRESHOLD_DEG_S,
395    gap_tolerance=GAP_TOLERANCE_S,
396    max_angle=MAX_CROSS_ANGLE_DEG,
397):
398    """
399    Run the full hold-detection chain on a per-frame deviation table.
400
401    Parameters
402    ----------
403    series : pandas.DataFrame
404        One row per frame with at least ``time_sec`` (s),
405        ``left_deviation`` and ``right_deviation`` (deg, NaN where no pose
406        was detected).
407    window_length : int, optional
408        Savitzky-Golay window in frames. None (default) chooses one with
409        `choose_window_length`.
410    polyorder : int, optional
411        Savitzky-Golay polynomial order. Default 2.
412    velocity_threshold : float, optional
413        Stop threshold in deg/s. Default 15.
414    gap_tolerance : float, optional
415        Largest gap in seconds bridged when merging segments. Default 0.25.
416    max_angle : float, optional
417        Largest mean deviation in degrees accepted as a cross. Default 45.
418
419    Returns
420    -------
421    frames : pandas.DataFrame
422        `series` with the smoothed angles, velocities and ``is_stopped``
423        flag added.
424    hold : pandas.Series or None
425        The detected hold (``start``, ``end``, ``duration``, ``avg_angle``),
426        or None if no plausible hold was found.
427
428    Raises
429    ------
430    KeyError
431        If a required column is missing.
432    ValueError
433        If the clip has fewer than `MIN_WINDOW_LENGTH` frames.
434    """
435    _check_columns(series, REQUIRED_COLUMNS)
436    if window_length is None:
437        window_length = choose_window_length(len(series))
438
439    frames = smooth_deviations(series, window_length, polyorder)
440    frames = add_angular_velocity(frames)
441    frames = flag_stopped_frames(frames, velocity_threshold)
442
443    segments = find_stopped_segments(frames)
444    merged = merge_segments(segments, gap_tolerance)
445    candidates = build_candidate_holds(frames, merged)
446    hold = select_main_hold(candidates, max_angle)
447    return frames, hold

450def judge_hold(hold):
451    """
452    Turn a detected hold into a judging summary.
453
454    Parameters
455    ----------
456    hold : pandas.Series or None
457        Output of `select_main_hold` or `detect_hold`.
458
459    Returns
460    -------
461    dict
462        ``start``, ``end``, ``duration``, ``avg_angle``, ``angle_deduction``
463        and ``duration_deduction``. If `hold` is None every field is None:
464        a missing hold can mean the athlete never stopped or that detection
465        failed, and the pipeline cannot tell which, so it does not score it.
466    """
467    if hold is None:
468        return dict.fromkeys(
469            ["start", "end", "duration", "avg_angle", "angle_deduction", "duration_deduction"]
470        )
471    return {
472        "start": float(hold["start"]),
473        "end": float(hold["end"]),
474        "duration": float(hold["duration"]),
475        "avg_angle": float(hold["avg_angle"]),
476        "angle_deduction": classify_deduction(hold["avg_angle"]),
477        "duration_deduction": classify_hold_duration(hold["duration"]),
478    }

481def timing_errors(hold, manual_start_frame, manual_end_frame, fps):
482    """
483    Compare a detected hold with a manually annotated one.
484
485    Parameters
486    ----------
487    hold : pandas.Series

237    """
238    Merge stopped segments separated by short gaps.
239
240    Parameters
241    ----------
242    segments : pandas.DataFrame
243        Output of `find_stopped_segments`, sorted by ``start_time``.
244    gap_tolerance : float, optional
245        Largest gap in seconds, from the end of one segment to the start of
246        the next, that is bridged. Default 0.25.
247
248    Returns
249    -------
250    list of tuple of float
251        ``(start_time, end_time)`` pairs in seconds, in time order. Empty if
252        `segments` is empty.
253
254    Examples
255    --------
256    >>> segs = pd.DataFrame({"start_time": [0.0, 1.1, 3.0], "end_time": [1.0, 2.0, 3.5]})

258    [(0.0, 2.0), (3.0, 3.5)]
259    """
260    merged = []
261    cur_start = cur_end = None
262    for _, row in segments.iterrows():
263        if cur_start is None:
264            cur_start, cur_end = row["start_time"], row["end_time"]
265        elif row["start_time"] - cur_end <= gap_tolerance:
266            cur_end = row["end_time"]
267        else:
268            merged.append((float(cur_start), float(cur_end)))
269            cur_start, cur_end = row["start_time"], row["end_time"]
270    if cur_start is not None:
271        merged.append((float(cur_start), float(cur_end)))
272    return merged
